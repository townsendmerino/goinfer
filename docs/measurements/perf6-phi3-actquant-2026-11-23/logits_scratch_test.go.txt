//go:build cuda && goinfer_testhooks

package cuda

import (
	"crypto/sha256"
	"encoding/binary"
	"fmt"
	"math"
	"os"
	"testing"

	"github.com/townsendmerino/goinfer/decoder"
)

// TestZZPerf6LogitsHash decodes PERF6_MODEL at q4k: 128 positions of token 1000, then 32 greedy tokens, and prints a SHA-256 over
// every logit of the 32 decode steps and the greedy ids. Scratch instrument for docs/measurements/perf6-phi3-actquant-2026-11-23,
// built into both arms (as zz_perf6_logits_test.go) so their hashes compare; not committed as a test.
func TestZZPerf6LogitsHash(t *testing.T) {
	path := os.Getenv("PERF6_MODEL")
	if path == "" {
		t.Skip("set PERF6_MODEL")
	}
	m, err := decoder.Load(path, decoder.Options{Backend: "cuda", Quant: "q4k", ResidentContext: 256})
	if err != nil {
		t.Fatal(err)
	}
	defer m.Close()
	rf := m.ResidentForwardForTest()
	if rf == nil {
		t.Fatalf("not resident: %s", m.ResidentDecline())
	}
	var lg []float32
	for pos := range 128 {
		if lg, err = rf.Forward(m.EmbedResidentForTest(1000), pos); err != nil {
			t.Fatal(err)
		}
	}
	h := sha256.New()
	var ids []int
	for pos := 128; pos < 160; pos++ {
		next := 0
		for i, v := range lg {
			if v > lg[next] {
				next = i
			}
		}
		ids = append(ids, next)
		if lg, err = rf.Forward(m.EmbedResidentForTest(next), pos); err != nil {
			t.Fatal(err)
		}
		var b [4]byte
		for _, v := range lg {
			binary.LittleEndian.PutUint32(b[:], math.Float32bits(v))
			h.Write(b[:])
		}
	}
	fmt.Fprintf(os.Stderr, "perf6-logits sha256 %x ids %v\n", h.Sum(nil), ids)
}
