//go:build darwin

package visionmetal

import (
	"math"
	"os"
	"testing"

	"github.com/townsendmerino/aikit/vision"
)

func TestZZSynthProbe(t *testing.T) {
	dir := os.Getenv("ZZ_SIGLIP")
	if dir == "" {
		t.Skip("ZZ_SIGLIP")
	}
	cpu, err := vision.LoadEncoder(dir, true)
	if err != nil {
		t.Fatal(err)
	}
	px := synthPixels(cpu)
	want, err := cpu.Forward(px)
	if err != nil {
		t.Fatal(err)
	}
	g, _ := vision.LoadEncoder(dir, true)
	if err := g.EnableResident(); err != nil {
		t.Fatal(err)
	}
	got, err := g.Forward(px)
	if err != nil {
		t.Fatal(err)
	}
	hidden := cpu.Cfg.HiddenSize
	worst, wi := 2.0, 0
	for i := 0; i < len(want)/hidden; i++ {
		c := cosine(got[i*hidden:(i+1)*hidden], want[i*hidden:(i+1)*hidden])
		if c < worst {
			worst, wi = c, i
		}
	}
	md := 0.0
	for i := range want {
		md = math.Max(md, math.Abs(float64(got[i]-want[i])))
	}
	ref, _ := vision.LoadEncoder(dir, false)
	f32, err := ref.Forward(px)
	if err != nil {
		t.Fatal(err)
	}
	wr := func(a []float32) (float64, float64) {
		w := 2.0
		for i := 0; i < len(a)/hidden; i++ {
			w = math.Min(w, cosine(a[i*hidden:(i+1)*hidden], f32[i*hidden:(i+1)*hidden]))
		}
		return cosine(a, f32), w
	}
	cw, cr := wr(want)
	gw, gr := wr(got)
	t.Logf("against f32 CPU: int8 CPU whole %.6f worst row %.6f; int8 Metal whole %.6f worst row %.6f", cw, cr, gw, gr)
	t.Logf("%s: whole cosine %.9f, worst row %d cosine %.9f, max|d| %.3g", dir, cosine(got, want), wi, worst, md)
}
