//go:build gpu && goinfer_testhooks

package webgpu

// Held-out set 5 of SB2-vk (docs/measurements/sb2-vk-2026-10-31/README.md, "Held-out set 5"): G-VK1's probe (one WebGPU load, its
// ledger's buffer list, nvidia-smi's used memory before the load, after it and after a fill), plus, from the loaded model and after
// every reading, the buffer list the planner builds from the model alone and the webgpu plan's priced bytes at the load's context.
// One process per load, so a checkpoint can be deleted as soon as its load ends. Kept in the record as a .go.txt; run-heldout5.sh
// copies it in.

import (
	"fmt"
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"testing"

	"github.com/oliverbestmann/webgpu/wgpu"
	"github.com/townsendmerino/goinfer/backend/webgpu/internal/kernels"
	"github.com/townsendmerino/goinfer/decoder"
	"github.com/townsendmerino/goinfer/internal/runtime/mem"
)

func TestSB2VK5Probe(t *testing.T) {
	path, out := os.Getenv("SB2_VK1_MODEL"), os.Getenv("SB2_VK1_OUT")
	if path == "" || out == "" {
		t.Skip("set SB2_VK1_MODEL and SB2_VK1_OUT")
	}
	opts := decoder.Options{Quant: "int4"}
	if s := os.Getenv("SB2_VK1_CTX"); s != "" {
		n, err := strconv.Atoi(s)
		if err != nil {
			t.Fatal(err)
		}
		opts.ResidentContext = n
	}
	if os.Getenv("SB2_VK1_SSM") != "" {
		opts.Knobs = &decoder.Knobs{"GOINFER_SSM_RESIDENT": os.Getenv("SB2_VK1_SSM")}
	}
	usedBefore := gpuVRAMMiB(t)
	b0 := LiveBufferBytes()
	// Loaded directly, not through loadWebGPUResident, whose fixture guard skips a directory without one model.safetensors (a sharded
	// checkpoint has an index).
	o := opts
	o.Backend, o.ResidentKVSlots = "webgpu", 1
	m, err := decoder.Load(path, o)
	if err != nil {
		t.Fatalf("load %s: %v", path, err)
	}
	defer m.Close()
	rd, ok := m.ResidentForwardForTest().(*residentDecoder)
	if !ok {
		t.Fatalf("%s did not go webgpu-resident: %s", path, m.ResidentDecline())
	}
	b := LiveBufferBytes() - b0
	bufs := rd.c.LeasedBuffersForTest()
	var ledger int64
	var list strings.Builder
	list.WriteString("seq\tbytes\tusage\tline\n")
	for _, x := range bufs {
		ledger += x.Bytes
		fmt.Fprintf(&list, "%d\t%d\t%#x\t%s\n", x.Seq, x.Bytes, x.Usage, x.Line)
	}
	usedLoaded := gpuVRAMMiB(t)
	var fill []*wgpu.Buffer
	var a int64
	fills := map[int64]int{}
	sizes := []int64{1 << 30, 256 << 20, 64 << 20, 16 << 20}
	if os.Getenv("SB2_VK5_NOFILL") != "" { // a smoke run on a box whose GPU shares host memory: no fill
		sizes = nil
	}
	for _, size := range sizes {
		for {
			buf, err := kernels.CreateBufferOn(rd.c, mem.LineScratch, &wgpu.BufferDescriptor{Label: "sb2-vk1-fill", Size: uint64(size), Usage: wgpu.BufferUsageStorage})
			if err != nil {
				break
			}
			fill = append(fill, buf)
			a += size
			fills[size>>20]++
		}
	}
	usedFull := gpuVRAMMiB(t)
	for _, buf := range fill {
		kernels.ReleaseBuffer(buf)
	}
	kv := m.KVDeliveredFor("webgpu")
	kvF16, kvI8 := kv == "f16", kv == "i8"
	s0, sk := vkPlanBuffers(m, rd.ctxCap, kvF16, kvI8)
	var planList int64
	for _, n := range s0 {
		planList += n
	}
	_, m1pList := vkSlots(1, s0, sk, 1<<62)
	p := m.Plan("webgpu", 1<<40, decoder.PlanRequest{Ctx: rd.ctxCap, CtxPinned: true, KVF16: kvF16, KVI8: kvI8})
	line := fmt.Sprintf("model=%s ctx=%d B=%d ledger_list=%d buffers=%d A=%d fills_MiB=%v nvsmi_used_MiB before=%d loaded=%d full=%d kv=%s plan_list=%d plan_list_buffers=%d m1p_plan_list=%d plan=%d plan_ctx=%d plan_placement=%v",
		filepath.Base(path), rd.ctxCap, b, ledger, len(bufs), a, fills, usedBefore, usedLoaded, usedFull, kv, planList, len(s0), m1pList, p.NeedBytes(), p.Ctx, p.Placement)
	if err := os.WriteFile(out, []byte(line+"\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(strings.TrimSuffix(out, ".txt")+"-buffers.tsv", []byte(list.String()), 0o644); err != nil {
		t.Fatal(err)
	}
	t.Log(line)
}
