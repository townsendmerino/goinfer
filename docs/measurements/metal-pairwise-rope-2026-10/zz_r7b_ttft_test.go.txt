//go:build darwin && goinfer_testhooks

package metal

import (
	"context"
	"fmt"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/townsendmerino/goinfer/decoder"
	"github.com/townsendmerino/goinfer/tokenizer"
)

// Throwaway (G-B4, exploratory): Command-R7B prefill of ~512 README tokens, batched against sequential, one run each.
func TestZZR7BTTFT(t *testing.T) {
	home, _ := os.UserHomeDir()
	dir := filepath.Join(home, "models", "command-r7b")
	tk, _ := tokenizer.Load(filepath.Join(dir, "tokenizer.json"))
	raw, _ := os.ReadFile("../README.md")
	ids, _ := tk.Encode(string(raw), false)
	ids = ids[:512]
	m, err := decoder.Load(dir+".int4.metal.giw", decoder.Options{Backend: "metal", Quant: "int4", ResidentContext: 1024})
	if err != nil {
		t.Fatal(err)
	}
	defer m.Close()
	a, ok := m.ResidentForwardForTest().(*metalResident)
	if !ok {
		t.Fatalf("not resident: %s", m.ResidentDecline())
	}
	embs := make([][]float32, len(ids))
	for i, id := range ids {
		embs[i] = m.EmbedResidentForTest(id)
	}
	a.Reset()
	if _, err := a.PrefillLast(context.Background(), embs[:64], 0); err != nil { // warm the prefill pipelines
		t.Fatal(err)
	}
	a.Reset()
	t0 := time.Now()
	if _, err := a.PrefillLast(context.Background(), embs, 0); err != nil {
		t.Fatal(err)
	}
	tb := time.Since(t0)
	a.Reset()
	t1 := time.Now()
	for i, e := range embs {
		if _, err := a.Forward(e, i); err != nil {
			t.Fatal(err)
		}
	}
	ts := time.Since(t1)
	fmt.Fprintf(os.Stderr, "[r7b %d tok] batched %.2fs, sequential %.2fs, speedup %.2fx\n", len(ids), tb.Seconds(), ts.Seconds(), ts.Seconds()/tb.Seconds())
}
