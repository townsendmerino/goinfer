//go:build gpu && goinfer_testhooks

package gpu

// G-WS3 of SB2 memory reads step 4 (docs/tasks/task-subsystems-2026-10.md): a WebGPU build's live device bytes (B), then what Vulkan
// still allocates beside it (A), filled with buffers of 1 GiB, 256, 64 and 16 MiB until each size fails. Kept in the record as a .go.txt.

import (
	"fmt"
	"os"
	"path/filepath"
	"testing"

	"github.com/oliverbestmann/webgpu/wgpu"
	"github.com/townsendmerino/goinfer/decoder"
	"github.com/townsendmerino/goinfer/gpu/internal/kernels"
	"github.com/townsendmerino/goinfer/internal/mem"
)

func TestSB2WS3Fill(t *testing.T) {
	path, out := os.Getenv("SB2_WS3_MODEL"), os.Getenv("SB2_WS3_OUT")
	if path == "" || out == "" {
		t.Skip("set SB2_WS3_MODEL and SB2_WS3_OUT")
	}
	usedBefore := gpuVRAMMiB(t)
	b0 := LiveBufferBytes()
	m, rd := loadWebGPUResident(t, path, decoder.Options{Quant: "int4"}, 1)
	defer m.Close()
	b := LiveBufferBytes() - b0
	usedLoaded := gpuVRAMMiB(t)
	var bufs []*wgpu.Buffer
	var a int64
	fills := map[int64]int{}
	for _, size := range []int64{1 << 30, 256 << 20, 64 << 20, 16 << 20} {
		for {
			buf, err := kernels.CreateBufferOn(rd.c, mem.LineScratch, &wgpu.BufferDescriptor{Label: "sb2-ws3-fill", Size: uint64(size), Usage: wgpu.BufferUsageStorage})
			if err != nil {
				break
			}
			bufs = append(bufs, buf)
			a += size
			fills[size>>20]++
		}
	}
	usedFull := gpuVRAMMiB(t)
	for _, buf := range bufs {
		kernels.ReleaseBuffer(buf)
	}
	line := fmt.Sprintf("model=%s ctx=%d B=%d A=%d fills_MiB=%v nvsmi_used_MiB before=%d loaded=%d full=%d",
		filepath.Base(path), rd.ctxCap, b, a, fills, usedBefore, usedLoaded, usedFull)
	if err := os.WriteFile(out, []byte(line+"\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	t.Log(line)
}
