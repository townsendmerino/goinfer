+//go:build !arm64
+
+package linalg
+
+import "unsafe"
+
+// dotF32I8 is dotF32(a, widen(q)) without the widened row: the same reduction,
+// reading q as float32(q[k]) in place. Bit-identical to dotF32 over
+// dequantRowInt8(·, q, 1.0), since that widen is exact and every add happens in
+// the same order (dotFMAI8 on AVX2, dotGenericI8 otherwise, then the same tail).
+func dotF32I8(a []float32, q []int8) float32 {
+	n := len(a)
+	if n != len(q) {
+		panic("linalg: dotF32I8 length mismatch")
+	}
+	if n == 0 {
+		return 0
+	}
+	n4 := n / 4
+	var sum float32
+	if n4 > 0 {
+		sum = dotI8Widen(&a[0], &q[0], n4)
+	}
+	for k := n4 * 4; k < n; k++ {
+		sum += a[k] * float32(q[k])
+	}
+	return sum
+}
+
+// dotGenericI8 is dotGeneric with b read as float32(q[i]).
+func dotGenericI8(a *float32, q *int8, n4 int) float32 {
+	n := n4 * 4
+	aSlice := unsafe.Slice(a, n)
+	qSlice := unsafe.Slice(q, n)
+	var sum float32
+	for i := range n {
+		sum += aSlice[i] * float32(qSlice[i])
+	}
+	return sum
+}

+// dotFMAI8 — Σ a[i]*float32(q[i]) over n elements (n a multiple of 4), the
+// int8 widen fused into dotFMA's loop. Every instruction that touches the
+// accumulators is dotFMA's, in dotFMA's order: four YMM accumulators over 32
+// elements, one over 8, the same fold, then the scalar tail as a VMULSS and a
+// VADDSS. VPMOVSXBD+VCVTDQ2PS gives exactly float32(q[i]), the value dotFMA
+// reads from the widened row, so the result is bit-identical to
+// dotFMA(a, widen(q), n).
+
+#include "textflag.h"
+
+// func dotFMAI8(a *float32, q *int8, n int) float32
+TEXT ·dotFMAI8(SB), NOSPLIT, $0-28
+	MOVQ a+0(FP), SI
+	MOVQ q+8(FP), DI
+	MOVQ n+16(FP), AX
+
+	VXORPS Y0, Y0, Y0
+	VXORPS Y1, Y1, Y1
+	VXORPS Y2, Y2, Y2
+	VXORPS Y3, Y3, Y3
+
+loop32:
+	CMPQ      AX, $32
+	JL        loop8
+	VMOVUPS   0(SI), Y4
+	VMOVUPS   32(SI), Y5
+	VMOVUPS   64(SI), Y6
+	VMOVUPS   96(SI), Y7
+	VPMOVSXBD 0(DI), Y8
+	VPMOVSXBD 8(DI), Y9
+	VPMOVSXBD 16(DI), Y10
+	VPMOVSXBD 24(DI), Y11
+	VCVTDQ2PS Y8, Y8
+	VCVTDQ2PS Y9, Y9
+	VCVTDQ2PS Y10, Y10
+	VCVTDQ2PS Y11, Y11
+	VFMADD231PS Y8, Y4, Y0
+	VFMADD231PS Y9, Y5, Y1
+	VFMADD231PS Y10, Y6, Y2
+	VFMADD231PS Y11, Y7, Y3
+	ADDQ      $128, SI
+	ADDQ      $32, DI
+	SUBQ      $32, AX
+	JMP       loop32
+
+loop8:
+	CMPQ        AX, $8
+	JL          reduce
+	VMOVUPS     0(SI), Y4
+	VPMOVSXBD   0(DI), Y8
+	VCVTDQ2PS   Y8, Y8
+	VFMADD231PS Y8, Y4, Y0
+	ADDQ        $32, SI
+	ADDQ        $8, DI
+	SUBQ        $8, AX
+	JMP         loop8
+
+reduce:
+	VADDPS       Y1, Y0, Y0
+	VADDPS       Y3, Y2, Y2
+	VADDPS       Y2, Y0, Y0
+	VEXTRACTF128 $1, Y0, X1
+	VADDPS       X1, X0, X0
+	VHADDPS      X0, X0, X0
+	VHADDPS      X0, X0, X0
+
+tail:
+	CMPQ       AX, $0
+	JE         done
+	MOVBLSX    0(DI), BX
+	VCVTSI2SSL BX, X2, X2
+	VMULSS     0(SI), X2, X2
+	VADDSS     X2, X0, X0
+	ADDQ       $4, SI
+	ADDQ       $1, DI
+	DECQ       AX
+	JMP        tail
+
+done:
+	VZEROUPPER
+	MOVSS X0, ret+24(FP)
+	RET

+// dotF32I8 against dotF32 over the widened row, raw bits, at every length to 300
+// (each %32, %8 and %4 tail) and at transformer K, with activations spread over
+// many binades so a reordered add would show.
+func TestDotF32I8_bitIdenticalToWidenThenDot(t *testing.T) {
+	rng := rand.New(rand.NewSource(0xd07))
+	lens := []int{}
+	for n := 0; n <= 300; n++ {
+		lens = append(lens, n)
+	}
+	lens = append(lens, 1023, 1536, 2048, 4095, 4096, 14336, 14339)
+	for _, n := range lens {
+		for trial := range 3 {
+			a := make([]float32, n)
+			q := make([]int8, n)
+			for i := range a {
+				a[i] = float32(rng.NormFloat64()) * float32(int(1)<<rng.Intn(20)) / 1024
+				q[i] = int8(rng.Intn(256) - 128)
+			}
+			deq := make([]float32, n)
+			dequantRowInt8(deq, q, 1.0)
+			want := dotF32(a, deq)
+			got := dotF32I8(a, q)
+			compareBits(t, fmt.Sprintf("n=%d trial=%d", n, trial), []float32{got}, []float32{want})
+		}
+	}
+}
