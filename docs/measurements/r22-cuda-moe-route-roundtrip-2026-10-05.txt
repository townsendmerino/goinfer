#!/bin/bash
# Exploratory A/B, R-22: old (21bf7ad8) vs new cuda test binary, real gemma4-26b (M26), M=512 batched prefill, ms/token. Order-balanced: ON NO ON... see ORDER.
S=$S
OUT=$S/ab-results.txt; : > $OUT
export GOINFER_HEAVY_TESTS=1 GOINFER_LONGPROMPT_MODEL=gemma4-26b-int4.giw GOINFER_LONGPROMPT_MOECACHE=1 GOINFER_LONGPROMPT_GIW=1 GOINFER_LONGPROMPT_LENGTHS=512
ORDER="old new new old old new"
i=0
t_start=$(date +%s)
for arm in $ORDER; do
  i=$((i+1))
  if [ $arm = old ]; then dir=$S/ab-wt/cuda; bin=$S/cuda-old.test; else dir=/home/francis/mycode/goinfer/cuda; bin=$S/cuda-new.test; fi
  log=$S/ab-run$i-$arm.log
  ( cd $dir && exec $bin -test.run TestPrefillLongPrompt -test.v -test.timeout 20m > $log 2>&1 ) &
  pid=$!
  for s in $(seq 1 300); do
    if grep -q "M=512 .*batched OK" $log 2>/dev/null || ! kill -0 $pid 2>/dev/null; then break; fi
    sleep 1
  done
  kill $pid 2>/dev/null; wait $pid 2>/dev/null
  line=$(grep "M=512 .*batched OK" $log | head -1)
  echo "run $i $arm: ${line:-NO RESULT}" | tee -a $OUT
  echo "[ab] $i/6 done, elapsed $(( $(date +%s) - t_start ))s"
  sleep 5
done
echo "[ab] finished"
